import { Steps, Tabs, TabItem } from '@astrojs/starlight/components';

<Steps>

1. **Install packages**

   <Tabs syncKey="framework">
     <TabItem label="React">
     ```bash
     npm install @maelstrom-co/react @maelstrom-co/client
     ```
     </TabItem>
     <TabItem label="Vanilla JS">
     ```bash
     npm install @maelstrom-co/client
     ```
     </TabItem>
   </Tabs>

2. **Define a Vessel**

   <Tabs syncKey="framework">
     <TabItem label="React">
     ```tsx title="vessels/greeting.tsx"
     import { vessel } from '@maelstrom-co/react';
     import { defineVessel } from '@maelstrom-co/protocol';

     const greetingDefinition = defineVessel({
       id: 'greeting',
       name: 'Greeting',
       description: 'Displays a greeting message',
       defaultVariant: 'compact',
       variants: {
         compact: {
           description: 'Short greeting',
           props: { message: 'Hello!' },
           minSize: { width: 2, height: 1 },
           actions: [],
         },
       },
     });

     function GreetingComponent({ variantProps }) {
       return <div>{variantProps.message}</div>;
     }

     export const greetingVessel = vessel(greetingDefinition, GreetingComponent);
     ```
     </TabItem>
     <TabItem label="Vanilla JS">
     ```ts title="vessels/greeting.ts"
     import type { Vessel } from '@maelstrom-co/protocol';

     export const greetingDefinition: Vessel = {
       id: 'greeting',
       name: 'Greeting',
       description: 'Displays a greeting message',
       defaultVariant: 'compact',
       variants: {
         compact: {
           description: 'Short greeting',
           props: { message: 'Hello!' },
           minSize: { width: 2, height: 1 },
           actions: [],
         },
       },
     };

     export function renderGreeting(container: HTMLElement, props: { message: string }) {
       container.textContent = props.message;
     }
     ```
     </TabItem>
   </Tabs>

3. **Create the app**

   <Tabs syncKey="framework">
     <TabItem label="React">
     ```tsx title="App.tsx"
     import { createMaelstrom } from '@maelstrom-co/react';
     import { greetingVessel } from './vessels/greeting';

     const m = createMaelstrom({
       vessels: [greetingVessel],
     });

     export function App({ connector }) {
       return (
         <m.Provider connector={connector}>
           <m.Chart className="h-screen" />
         </m.Provider>
       );
     }
     ```
     </TabItem>
     <TabItem label="Vanilla JS">
     ```ts title="main.ts"
     import { Orchestrator, WebSocketConnector } from '@maelstrom-co/client';
     import { greetingDefinition, renderGreeting } from './vessels/greeting';

     const connector = new WebSocketConnector({ url: 'wss://api.example.com/ws' });
     const orchestrator = new Orchestrator(connector);

     // Register vessel definitions
     orchestrator.vesselManager.registerVessel(greetingDefinition);

     // Set chart dimensions
     orchestrator.chartManager.setDimensions({ columns: 12, rows: 8 });

     // Connect and process user intent
     await orchestrator.connect();
     await orchestrator.processIntent('show me a greeting');
     ```
     </TabItem>
   </Tabs>

</Steps>
